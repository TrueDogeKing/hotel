        var total = FormatZl(booking.TotalGrosze);
        var deposit = FormatZl(booking.DepositGrosze);

        var stay = Stay(booking, booking.Language);

        if (booking.Language == "en")
        {
            return new EmailMessage(
                booking.Email,

                $"""

                {stay} ({booking.Nights} nights).

                    "en"
