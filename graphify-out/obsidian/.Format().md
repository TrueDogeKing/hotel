    private static string FormatDateTime(DateTime? utc, string language) =>
        utc is null
            ? "-"
            : utc.Value.ToString(
                "d MMMM yyyy HH:mm 'UTC'",
                new CultureInfo(language == "en" ? "en-GB" : "pl-PL")

    private static string FormatZl(long grosze) =>
