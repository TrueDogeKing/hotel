    [Fact]
    public void ValidateMix_RejectsRedundantRoom()
    {
        var free = new Dictionary<int, int> { [4] = 10, [2] = 10 };
        var counts = new Dictionary<int, int> { [4] = 3 }; // 12 beds for 5 people — one 4 is redundant

    }
